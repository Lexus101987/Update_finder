import requests
import shutil
import os
import sys

# 1. Configuration
CURRENT_VERSION = "1.0.0"
VERSION_URL = "https://raw.githubusercontent.com/YourUser/YourRepo/main/version.txt"
UPDATE_URL = "https://raw.githubusercontent.com/YourUser/YourRepo/main/my_script.py"

def update_system():
    try:
        # 2. Check Version
        response = requests.get(VERSION_URL)
        online_version = response.text.strip()

        if online_version > CURRENT_VERSION:
            print(f"New version {online_version} found. Downloading...")
            
            # 3. Download the new file
            new_file_data = requests.get(UPDATE_URL)
            
            # 4. Save the update to a temp file first (Safety First!)
            with open("my_script_new.py", "wb") as f:
                f.write(new_file_data.content)
            
            # 5. Swap the files
            # On next restart, the user runs the new version
            shutil.move("my_script_new.py", "my_script.py")
            
            print("Update complete! Please restart the program.")
            sys.exit() 
        else:
            print("You are up to date.")

    except Exception as e:
        print(f"Update failed: {e}")

if __name__ == "__main__":
    update_system()
    # Your actual program starts here
    print("Running App v" + CURRENT_VERSION)
